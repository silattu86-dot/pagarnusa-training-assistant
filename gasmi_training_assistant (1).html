2. SUPABASE SQL + ROW LEVEL SECURITY (RLS) 
-- Tabel Profil Pengguna (Akun Anggota dibuat oleh Pelatih via Admin Function)
create table profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  full_name text not null,
  role text not null check (role in ('pelatih', 'anggota')),
  age_group text check (age_group in ('anak', 'remaja')),
  skill_level text default 'pemula',
  updated_at timestamptz default now()
);
alter table profiles enable row level security;

-- Pelatih baca/kelola semua; Anggota hanya baca profil sendiri
create policy "Pelatih full access profiles" on profiles
  for all using (exists (select 1 from profiles p where p.id = auth.uid() and p.role = 'pelatih'));
create policy "Anggota read own profile" on profiles
  for select using (id = auth.uid());

-- Tabel Kurikulum 104 Pertemuan
create table curriculum_meetings (
  id int primary key,
  stage int not null,
  theme text not null,
  objective text not null,
  main_material text not null,
  movement_material text not null,
  warmup text not null,
  core_drill text not null,
  cooldown text not null,
  evaluation_guide text not null,
  coach_notes text,
  verified_by uuid references profiles(id),
  updated_at timestamptz default now()
);
alter table curriculum_meetings enable row level security;
create policy "Semua user baca kurikulum" on curriculum_meetings for select using (auth.role() = 'authenticated');
create policy "Hanya pelatih ubah kurikulum" on curriculum_meetings for all using (
  exists (select 1 from profiles p where p.id = auth.uid() and p.role = 'pelatih')
);

-- Tabel Presensi & Evaluasi (Last-Write-Wins via updated_at)
create table attendance_records (
  id uuid primary key,
  meeting_id int references curriculum_meetings(id),
  member_id uuid references profiles(id),
  session_date date not null,
  status text check (status in ('Hadir','Izin','Sakit','Alpa')),
  updated_at timestamptz default now()
);
alter table attendance_records enable row level security;
create policy "Pelatih kelola presensi" on attendance_records for all using (
  exists (select 1 from profiles p where p.id = auth.uid() and p.role = 'pelatih')
);
create policy "Anggota baca presensi sendiri" on attendance_records for select using (member_id = auth.uid());
3. EDGE FUNCTION AI AMAN (supabase/functions/gasmi-ai/index.ts) 
import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
// API KEY disimpan di Supabase Secrets (AI_API_KEY), TIDAK PERNAH di APK Android
const SYSTEM_PROMPT = `Anda adalah GASMI AI ASSISTANT untuk Pelatih Pagar Nusa GASMI Ranting Nguruan.
ATURAN MUTLAK:
1. Anda asisten pelatih, BUKAN pengganti pelatih.
2. DILARANG mengarang jurus/teknik dan mengklaimnya sebagai materi resmi GASMI.
3. Hanya gunakan referensi materi terverifikasi yang dikirim dalam konteks.
4. Utamakan keselamatan anak-anak dan remaja (tanpa sparring keras untuk pemula).
5. DILARANG memberi diagnosis medis atau memaksa latihan saat cedera.`;

serve(async (req) => {
  const { prompt, verifiedMaterials, ageGroup } = await req.json();
  const apiKey = Deno.env.get("AI_API_KEY");
  const response = await fetch("https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=" + apiKey, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      contents: [{ parts: [{ text: `${SYSTEM_PROMPT}\nKelompok Usia: ${ageGroup}\nMateri Terverifikasi: ${JSON.stringify(verifiedMaterials)}\nPermintaan Pelatih: ${prompt}` }] }]
    })
  });
  const data = await response.json();
  return new Response(JSON.stringify(data), { headers: { "Content-Type": "application/json" } });
});
